✅連線成功並已完成首次雲端同步！您的 Google 試算表已建立「庫存總表」、「異動紀錄」與「登入帳號」三個分頁。建議接著點擊下方「下載已綁定雲端網址的 index.html」上傳至 GitHub，永久保留此連線設定。
請將以下完整程式碼貼入 Google 試算表的「擴充功能 → Apps Script」編輯器並部署為「網頁應用程式（所有人）」： 
function doGet(e) {
  return handleRequest('GET', null);
}

function doPost(e) {
  var payload = {};
  if (e && e.postData && e.postData.contents) {
    payload = JSON.parse(e.postData.contents);
  }
  return handleRequest('POST', payload);
}

function handleRequest(method, payload) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var itemSheet = getOrCreateSheet(ss, '庫存總表', ['品項ID', '物品名稱', '現有庫存', '最後更新時間']);
    var logSheet = getOrCreateSheet(ss, '異動紀錄', ['紀錄ID', '異動時間', '類型', '物品名稱', '異動數量', '異動後結餘', '登記人員', '領用人與部門', '事由備註']);
    var userSheet = getOrCreateSheet(ss, '登入帳號', ['帳號', '密碼', '權限類型']);

    if (method === 'POST' && payload && payload.action === 'syncAll') {
      writeRows(itemSheet, ['品項ID', '物品名稱', '現有庫存', '最後更新時間'], (payload.items || []).map(function(i) {
        return [i.id, i.name, Number(i.qty || 0), i.updatedAt || ''];
      }));
      writeRows(logSheet, ['紀錄ID', '異動時間', '類型', '物品名稱', '異動數量', '異動後結餘', '登記人員', '領用人與部門', '事由備註'], (payload.logs || []).map(function(l) {
        return [l.id, l.timestamp, l.type === 'in' ? '入庫' : '出庫', l.itemName, Number(l.qty || 0), Number(l.afterQty || 0), l.operator || '', l.recipient || '', l.note || ''];
      }));
      writeRows(userSheet, ['帳號', '密碼', '權限類型'], (payload.users || []).map(function(u) {
        return [u.username, u.password, u.isDefault ? '預設管理員' : '一般使用者'];
      }));
    }

    var items = readSheetData(itemSheet).map(function(r) {
      return { id: String(r[0]), name: String(r[1]), qty: Number(r[2] || 0), updatedAt: String(r[3] || '') };
    });
    var logs = readSheetData(logSheet).map(function(r) {
      return {
        id: String(r[0]),
        timestamp: String(r[1]),
        type: r[2] === '入庫' ? 'in' : 'out',
        itemName: String(r[3]),
        qty: Number(r[4] || 0),
        afterQty: Number(r[5] || 0),
        operator: String(r[6] || ''),
        recipient: String(r[7] || ''),
        note: String(r[8] || '')
      };
    });
    var users = readSheetData(userSheet).map(function(r) {
      return { username: String(r[0]), password: String(r[1]), isDefault: r[2] === '預設管理員' };
    });

    return ContentService.createTextOutput(JSON.stringify({
      status: 'ok',
      items: items,
      logs: logs,
      users: users
    })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function getOrCreateSheet(ss, name, headers) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
  }
  return sheet;
}

function writeRows(sheet, headers, rows) {
  sheet.clearContents();
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
  if (rows.length > 0) {
    sheet.getRange(2, 1, rows.length, headers.length).setValues(rows);
  }
}

function readSheetData(sheet) {
  var lastRow = sheet.getLastRow();
  var lastCol = sheet.getLastColumn();
  if (lastRow <= 1 || lastCol === 0) return [];
  return sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();
}
